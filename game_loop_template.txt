		while(game_loop_running && !game_paused)
		{
			// Cont. time-to-update logic
			current_time = System.nanoTime();
			delta += (current_time - last_time) / draw_interval;
			last_time = current_time;
			if(delta >= 1)

				update();

				delta = 0;
